cmake_minimum_required(VERSION 3.24)
project(ArchiOfficeDriveWindows LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The CFAPI provider requires the Windows SDK")
endif()
add_executable(archioffice-drive-cfapi provider.cpp)
target_compile_features(archioffice-drive-cfapi PRIVATE cxx_std_20)
target_compile_definitions(archioffice-drive-cfapi PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(archioffice-drive-cfapi PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(archioffice-drive-cfapi PRIVATE CldApi windowsapp)
set_property(TARGET archioffice-drive-cfapi PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
