cmake_minimum_required(VERSION 3.24)
project(ArchiOfficeShell LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "The Explorer extension requires Windows")
endif()
add_library(archioffice-drive-shell SHARED dll.cpp folder.cpp view.cpp ipc.cpp shell.def)
set(SHELL_BUILD_ID "dev" CACHE STRING "Versioned DLL identity")
if(NOT SHELL_BUILD_ID MATCHES "^[a-zA-Z0-9-]+$")
 message(FATAL_ERROR "Invalid shell build identity")
endif()
set_target_properties(archioffice-drive-shell PROPERTIES OUTPUT_NAME "archioffice-drive-shell-${SHELL_BUILD_ID}")
configure_file(shell.rc.in shell.rc @ONLY)
add_executable(archioffice-shell-control control.cpp ${CMAKE_CURRENT_BINARY_DIR}/shell.rc)
target_compile_definitions(archioffice-shell-control PRIVATE SHELL_DLL_FILE=L\"archioffice-drive-shell-${SHELL_BUILD_ID}.dll\")
foreach(target archioffice-drive-shell archioffice-shell-control)
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
 target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
 target_link_libraries(${target} PRIVATE windowsapp Shell32 Ole32 Shlwapi Comctl32 Uxtheme Advapi32 Gdiplus)
endforeach()
