cmake_minimum_required(VERSION 3.24)
project(ArchiOfficeDriveShell LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The Shell prototype requires Windows")
endif()
add_library(archioffice-drive-shell SHARED shell.cpp exports.def)
target_compile_features(archioffice-drive-shell PRIVATE cxx_std_20)
target_compile_definitions(archioffice-drive-shell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(archioffice-drive-shell PRIVATE /W4 /EHsc /utf-8)
  set_property(TARGET archioffice-drive-shell PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(archioffice-drive-shell PRIVATE -static-libgcc -static-libstdc++)
endif()
target_link_libraries(archioffice-drive-shell PRIVATE shell32 shlwapi ole32 uuid comctl32 windowsapp advapi32 gdi32 user32)
set_target_properties(archioffice-drive-shell PROPERTIES PREFIX "")
add_executable(archioffice-shell-smoke smoke.cpp)
target_compile_features(archioffice-shell-smoke PRIVATE cxx_std_20)
target_compile_definitions(archioffice-shell-smoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(archioffice-shell-smoke PRIVATE shell32 ole32 uuid comctl32 user32 windowsapp)
if(MSVC)
  target_compile_options(archioffice-shell-smoke PRIVATE /W4 /EHsc /utf-8)
endif()
